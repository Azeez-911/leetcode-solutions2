// LeetCode #184 - Department Highest Salary
// Difficulty : Medium
// Language   : unknown
// Runtime    : 992 ms
// Memory     : N/A
// URL        : https://leetcode.com/problems/department-highest-salary/

SELECT
    d.name AS Department,
    e.name AS Employee,
    e.salary AS Salary
FROM Employee e
JOIN Department d
    ON e.departmentId = d.id
WHERE e.salary = (
    SELECT MAX(e2.salary)
    FROM Employee e2
    WHERE e2.departmentId = e.departmentId
);
